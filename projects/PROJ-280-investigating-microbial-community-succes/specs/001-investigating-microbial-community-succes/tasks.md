- [ ] T004_script [P] Create `code/validators.py` with a fully runnable `validate_dataset_config` function:

```python
import json
import re
import sys
import yaml
from jsonschema import validate, ValidationError

VALID_SRA = re.compile(r'^(SRR|ERR)[0-9]+$')
VALID_ZENDO = re.compile(r'^10\.5281/zenodo\.[0-9]+$')

def validate_dataset_config(config_path: str) -> bool:
 try:
 with open(config_path, 'r') as f:
 data = json.load(f)
 with open('contracts/dataset-config.schema.yaml', 'r') as sf:
 schema = yaml.safe_load(sf)
 validate(instance=data, schema=schema)
 for ds in data['datasets']:
 if ds['source'] == 'NCBI_SRA' and not VALID_SRA.match(ds['id']):
 raise ValueError(f"Invalid SRA ID: {ds['id']}")
 if ds['source'] == 'Zenodo' and not VALID_ZENDO.match(ds['url']):
 raise ValueError(f"Invalid Zenodo URL: {ds['url']}")
 return True
 except (ValidationError, ValueError, json.JSONDecodeError) as e:
 print(f"Validation failed: {e}", file=sys.stderr)
 return False
```
